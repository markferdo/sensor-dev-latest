def run_serial_bridge():
    keys_sent = False
    
    while True:
        try:
            print(f"--- Attempting to connect to /dev/ttyACM* ---")
            ser = serial.Serial('/dev/ttyACM0', 115200, timeout=0.1) # change the port
            ser.dsrdtr = False
            ser.rtscts = False
            ser.setDTR(False)
            ser.setRTS(False)

            print("--- Connected! Listening for logs... ---")

            while True:
                line = ser.readline().decode(errors='ignore').strip()
                if line:
                    print(f"[ESP]: {line}")

                    if "READY_TO_RECEIVE" in line and not keys_sent:
                        print(">>> Handshake! Sending Keys...")
                        time.sleep(0.5)
                        ser.write((json.dumps(data) + "\n").encode('utf-8'))
                        ser.flush()
                        keys_sent = True
                    
                    if "Read back:" in line:
                        print("\nSUCCESS!")

        except (serial.SerialException, OSError):
            print("--- Device disconnected/reset. Reconnecting in 1s... ---")
            time.sleep(1)
            continue
        except KeyboardInterrupt:
            print("\nExiting...")
            break

if __name__ == "__main__":
    run_serial_bridge()
